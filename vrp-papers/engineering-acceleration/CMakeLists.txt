cmake_minimum_required(VERSION 3.16)
project(bpc_engineering LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Solver selection: GUROBI (default) or COPT
set(SOLVER "GUROBI" CACHE STRING "LP solver backend: COPT or GUROBI")

if(SOLVER STREQUAL "GUROBI")
    if(DEFINED ENV{GUROBI_HOME})
        set(_GUROBI_DEFAULT "$ENV{GUROBI_HOME}")
    else()
        file(GLOB _gurobi_candidates "C:/gurobi*/win64")
        if(_gurobi_candidates)
            list(SORT _gurobi_candidates)
            list(GET _gurobi_candidates -1 _GUROBI_DEFAULT)
        endif()
    endif()
    set(GUROBI_HOME "${_GUROBI_DEFAULT}" CACHE PATH "Gurobi installation directory")

    add_compile_definitions(USE_GUROBI)
    include_directories("${GUROBI_HOME}/include")
    link_directories("${GUROBI_HOME}/lib")

    file(GLOB GUROBI_LIB "${GUROBI_HOME}/lib/gurobi[0-9]*.lib")
    file(GLOB GUROBI_CXX_LIB "${GUROBI_HOME}/lib/gurobi_c++md*.lib")
    if(NOT GUROBI_CXX_LIB)
        file(GLOB GUROBI_CXX_LIB "${GUROBI_HOME}/lib/gurobi_c++mt*.lib")
    endif()
    foreach(_lib ${GUROBI_CXX_LIB})
        get_filename_component(_name ${_lib} NAME_WE)
        if(NOT _name MATCHES "d2017$")
            set(GUROBI_CXX_PICK ${_lib})
            break()
        endif()
    endforeach()
    if(NOT GUROBI_CXX_PICK)
        list(GET GUROBI_CXX_LIB 0 GUROBI_CXX_PICK)
    endif()
    list(GET GUROBI_LIB 0 GUROBI_LIB)
    get_filename_component(GUROBI_LIB_NAME "${GUROBI_LIB}" NAME_WE)
    get_filename_component(GUROBI_CXX_LIB_NAME "${GUROBI_CXX_PICK}" NAME_WE)
    set(SOLVER_LIBS ${GUROBI_CXX_LIB_NAME} ${GUROBI_LIB_NAME})

elseif(SOLVER STREQUAL "COPT")
    set(COPT_HOME "C:/Program Files/copt70" CACHE PATH "COPT installation directory")
    add_compile_definitions(USE_COPT)
    include_directories("${COPT_HOME}/include" "${COPT_HOME}/include/coptcpp_inc")
    link_directories("${COPT_HOME}/lib")
    set(SOLVER_LIBS copt copt_cpp)
endif()

add_executable(bpc_solver main.cpp)
target_link_libraries(bpc_solver ${SOLVER_LIBS})
